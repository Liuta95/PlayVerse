export const colors = Object.entries(tokens).filter(([key, t]) => !key.startsWith('$') && t.$type === 'color');

# PlayVerse foundations

Tokens come from Figma Variables into `src/tokens/tokens.json` (W3C DTCG). `npm run tokens` generates `src/tokens/tokens.css` from it — never edit either by hand.

## Colors

<ColorPalette>
  {colors.map(([name, t]) => (
    <ColorItem key={name} title={name} subtitle={`--${name}`} colors={{ [name]: t.$value }} />
  ))}
</ColorPalette>
